# Sanitized BOM Validator Skill (illustrative, fictional data)
# Companion to portfolio project "Materials (BOM) Validator Skill".
# No internal systems referenced. Runs standalone with a mock inventory.

from dataclasses import dataclass

@dataclass
class Item:
    part: str
    kind: str          # patch_cord | optic | hardware
    length_m: int = 0
    qty_needed: int = 0

# Fictional inventory: part -> (description, units_available)
MOCK_INVENTORY = {
    "XX-000111": ("Patch cord 10 m", 60),
    "XX-000112": ("Patch cord 10 m (alt vendor)", 40),
    "XX-000223": ("Patch cord 30 m", 40),
    "XX-000334": ("100G optic module", 12),
}

def classify(part: str, length_m: int = 0) -> str:
    if length_m > 0:
        return "patch_cord"
    if part.startswith("XX-0003"):
        return "optic"
    return "hardware"

def check_stock(part: str, qty: int):
    desc, avail = MOCK_INVENTORY.get(part, ("unknown", 0))
    return {"part": part, "description": desc, "needed": qty,
            "available": avail, "ok": avail >= qty}

def find_substitute(length_m: int, qty: int):
    """Find any patch cord of the same length with enough stock."""
    for part, (desc, avail) in MOCK_INVENTORY.items():
        if f"{length_m} m" in desc and avail >= qty:
            return {"substitute": part, "description": desc,
                    "available": avail}
    return None

def validate_bom(items: list[dict]) -> list[dict]:
    results = []
    for it in items:
        kind = classify(it["part"], it.get("length_m", 0))
        r = check_stock(it["part"], it["qty"])
        r["kind"] = kind
        if not r["ok"] and kind == "patch_cord":
            r["alternative"] = find_substitute(
                it.get("length_m", 0), it["qty"])
        results.append(r)
    return results

if __name__ == "__main__":
    sample = [
        {"part": "XX-000111", "length_m": 10, "qty": 24},
        {"part": "XX-000222", "length_m": 30, "qty": 12},
        {"part": "XX-000333", "qty": 8},
    ]
    for row in validate_bom(sample):
        print(row)

# Sanitized Shipment Tracker Skill (illustrative, fictional data)
# Companion to portfolio project "Hardware Kit Shipment Tracker".
# No internal systems referenced. Runs standalone with mock data.

from dataclasses import dataclass

@dataclass
class Shipment:
    kit: str
    carrier: str
    tracking_no: str
    status: str        # IN_TRANSIT | DELIVERED | PENDING | EXCEPTION
    eta: str

# Public carrier tracking pages (official public URLs)
CARRIER_URLS = {
    "fedex": "https://www.fedex.com/fedextrack/?trknbr=",
    "ups": "https://www.ups.com/track?tracknum=",
    "dhl": "https://www.dhl.com/global-en/home/tracking/"
           "tracking-express.html?submit=1&tracking-id=",
    "usps": "https://tools.usps.com/go/TrackConfirmAction?tLabels=",
}

MOCK_SHIPMENTS = {
    "KIT-22001": [Shipment("KIT-22001", "fedex",
                            "999000111222", "DELIVERED", "Mar 4")],
    "KIT-22002": [Shipment("KIT-22002", "ups",
                            "1Z9999999999999999", "IN_TRANSIT", "Mar 11"),
                  Shipment("KIT-22002", "dhl",
                            "8888888888", "PENDING", "Mar 14")],
}

def tracking_link(carrier: str, number: str) -> str:
    base = CARRIER_URLS.get(carrier.lower())
    return base + number if base else number

def track_kits(kits: list[str]) -> list[dict]:
    """Batch track; one unknown kit never fails the whole report."""
    report = []
    for kit in kits:
        for s in MOCK_SHIPMENTS.get(kit, []):
            report.append({"kit": s.kit, "carrier": s.carrier,
                           "tracking_no": s.tracking_no,
                           "status": s.status, "eta": s.eta,
                           "link": tracking_link(s.carrier,
                                                 s.tracking_no)})
        if kit not in MOCK_SHIPMENTS:
            report.append({"kit": kit, "status": "NOT_FOUND"})
    return report

if __name__ == "__main__":
    for row in track_kits(["KIT-22001", "KIT-22002", "KIT-99999"]):
        print(row)
How I work
